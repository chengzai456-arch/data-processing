### 场景2：修改排班正确判定逻辑

```typescript
export function rule2ShiftCorrect(rows, sd) {
  return rows.map((x) => {
    const sn = (x.shift_name || "").trim();  // 班次名称
    const fp = (x.first_punch || "").trim();  // 首打卡时间
    const lp = (x.last_punch || "").trim();   // 末打卡时间

    // P1: 没班次 -> "/"
    if (!sn) return { ...x, is_schedule_correct: "/" };
    // P3: 有班次没末打卡，看首打卡和上班时间差
    if (有班次且没末打卡) {
      const 差值 = 首打卡时间 - 班次上班时间;
      if (差值 <= 60分钟) return { ...x, is_schedule_correct: "正确" };  // 60就是阈值
      ...
    }
  });
}
```

**想改判定条件？** 直接改 `if` 里面的逻辑就行。比如把 60 分钟改成 90 分钟：把 `<= 60` 改成 `<= 90`。
